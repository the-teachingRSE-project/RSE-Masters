  % part box: \gsbox{x1}{y1}{x2}{y2}{colour}{text}
  \newcommand{\gsbox}[6]{%
    \draw[part,draw=#5] (#1,#2) rectangle (#3,#4);
    \node[align=center] at ($(#1,#2)!0.5!(#3,#4)$) {#6};}
  % chevron: \gschev{x}{width}{ytop}{ybottom}{text}
  \newcommand{\gschev}[5]{%
    \draw[part,draw=gsBlueD,rounded corners=1pt]
      (#1,#3) -- ++(#2-0.25,0) -- (#1+#2,{(#3+#4)/2}) -- (#1+#2-0.25,#4)
      -- (#1,#4) -- (#1+0.25,{(#3+#4)/2}) -- cycle;
    \node at (#1+#2/2+0.05,{(#3+#4)/2}) {#5};}
